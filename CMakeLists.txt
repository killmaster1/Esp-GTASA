cmake_minimum_required(VERSION 3.22)
project(npclines LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Configure with the Android NDK toolchain and arm64-v8a")
endif()
add_library(npclines SHARED main.cpp)
target_compile_features(npclines PRIVATE cxx_std_17)
target_include_directories(npclines PRIVATE .)
target_compile_options(npclines PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(npclines PRIVATE log dl)
