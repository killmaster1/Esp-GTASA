cmake_minimum_required(VERSION 3.22)
project(headcam LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use Android NDK toolchain with arm64-v8a")
endif()
add_library(headcam SHARED main.cpp)
target_compile_features(headcam PRIVATE cxx_std_17)
target_include_directories(headcam PRIVATE ..)
target_compile_options(headcam PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(headcam PRIVATE log dl)
