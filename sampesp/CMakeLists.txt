cmake_minimum_required(VERSION 3.22)
project(sampesp LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use Android NDK toolchain with arm64-v8a")
endif()
add_library(sampesp SHARED main.cpp)
target_compile_features(sampesp PRIVATE cxx_std_17)
target_include_directories(sampesp PRIVATE ..)
target_compile_options(sampesp PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(sampesp PRIVATE log dl)
